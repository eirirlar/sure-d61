% Sunlit Sea report theme — header-includes for pandoc + xelatex (article class).
% Companion to scripts/sunlit_sea_slides_header.tex but for long-form documents.
% Colors and typography follow the Sunlit Sea Visual Profile Overview.
%
% The font prelude (loading Everett .ttf from visual_profile/fonts/) is
% injected by scripts/md_to_sunlit_report.py via a prelude -H include that
% defines \everett and \everettMedium. A fallback is provided here so the
% compilation still succeeds if the wrapper is bypassed.

% --- Section headings: Everett, black, yellow underline accent --------
% Note: the yellow rule after the heading uses \textcolor (not \color) so
% the color scopes correctly around \rule — earlier \color{sunlitYellow}
% version did not render the rule in yellow.
\titleformat{\section}
  {\everett\color{sunlitBlack}\Large\bfseries}
  {\thesection.}{0.6em}{}
  [\vspace{2pt}\textcolor{sunlitYellow}{\rule{1.5em}{2.5pt}}]

\titleformat{\subsection}
  {\everett\color{sunlitBlack}\Large\bfseries}
  {\thesubsection}{0.5em}{}

\titleformat{\subsubsection}
  {\everett\color{sunlitBlack700}\normalsize\bfseries}
  {\thesubsubsection}{0.4em}{}

\titlespacing*{\section}{0pt}{2.4ex plus 0.4ex minus 0.2ex}{1.2ex}
\titlespacing*{\subsection}{0pt}{2.0ex plus 0.3ex minus 0.2ex}{0.8ex}
\titlespacing*{\subsubsection}{0pt}{1.0ex plus 0.2ex minus 0.2ex}{0.4ex}

% --- Orphan protection: pull short subsection stubs forward -----------
\pretocmd{\subsection}{\needspace{6\baselineskip}}{}{}
\pretocmd{\subsubsection}{\needspace{4\baselineskip}}{}{}

% --- Title block styling (overrides pandoc default) --------------------
\pretitle{\begin{flushleft}\everett\color{sunlitBlack}\Huge\bfseries}
\posttitle{\par\vspace{4pt}\textcolor{sunlitYellow}{\rule{3em}{3pt}}\par\end{flushleft}\vspace{1em}}
\preauthor{\begin{flushleft}\color{sunlitBlack700}\normalsize}
\postauthor{\par\end{flushleft}}
\predate{\begin{flushleft}\color{sunlitBlack700}\small}
\postdate{\par\end{flushleft}\vspace{1.5em}}

% --- Table of contents styling -----------------------------------------
\renewcommand{\contentsname}{\everett\color{sunlitBlack}\Large\bfseries Contents}
\setcounter{tocdepth}{2}
\titlecontents{section}
  [0em]
  {\addvspace{0.4em}\everett\color{sunlitBlack}\normalsize}
  {\contentslabel{1.8em}}
  {}
  {\titlerule*[0.6pc]{.}\contentspage}
\titlecontents{subsection}
  [2.2em]
  {\color{sunlitBlack700}\small}
  {\contentslabel{2.4em}}
  {}
  {\titlerule*[0.6pc]{.}\contentspage}

% --- Page headers / footers --------------------------------------------
\pagestyle{fancy}
