% EU emblem (12 gold stars on blue field, aspect ratio 3:2), drawn in TikZ
% to the geometry of surewave/images/eu_flag.svg (stars on a circle of
% radius 1/3 of the flag height). Usage: \euemblem{<width>}
\definecolor{eublue}{RGB}{0,51,153}
\definecolor{eustars}{RGB}{255,204,0}
\newcommand{\euemblem}[1]{%
\begin{tikzpicture}[x=#1/3,y=#1/3]
\fill[eublue] (0,0) rectangle (3,2);
\foreach \i in {0,...,11} {%
  \node[star, star points=5, star point ratio=2.618,
        fill=eustars, draw=none, inner sep=0,
        minimum size=#1/13.5] at ({1.5+0.6667*cos(\i*30+90)},{1+0.6667*sin(\i*30+90)}) {};%
}
\end{tikzpicture}%
}
