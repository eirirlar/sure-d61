% Pandoc header-includes for SUREWAVE newsletters (III, IV, V).
% Used via `pandoc --include-in-header=scripts/newsletter_pandoc_header.tex`.
% The issue label for the footer is set in the newsletter Markdown with
%   \renewcommand{\nlfooter}{SUREWAVE $\cdot$ Newsletter III $\cdot$ December 2026}
% before any text, so one header serves all issues.

\usepackage{float}
\floatplacement{figure}{!ht}

% Captions: small, numbered "Figure N:".
\usepackage{caption}
\captionsetup{font=small,labelfont=bf,labelsep=colon,width=.9\linewidth}

% Tighter paragraphs for a newsletter look.
\setlength{\parskip}{0.5em}
\setlength{\parindent}{0pt}
\widowpenalty=10000
\clubpenalty=10000

% Section headings in project blue.
\usepackage{titlesec}
\definecolor{swblue}{RGB}{25,25,112}
\titleformat{\section}{\Large\bfseries\color{swblue}}{}{0pt}{}
\titleformat{\subsection}{\large\bfseries\color{swblue}}{}{0pt}{}
\titlespacing*{\section}{0pt}{1.0em}{0.4em}
\titlespacing*{\subsection}{0pt}{0.8em}{0.3em}

% Running footer.
\providecommand{\nlfooter}{SUREWAVE newsletter}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\small\nlfooter \quad p.\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0.4pt}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\small\nlfooter \quad p.\thepage}\renewcommand{\headrulewidth}{0pt}\renewcommand{\footrulewidth}{0.4pt}}

% Shaded box for the project one-liner and the contact block.
\usepackage[most]{tcolorbox}
\newtcolorbox{nlbox}{colback=swblue!6,colframe=swblue!40,boxrule=0.4pt,arc=2pt,left=6pt,right=6pt,top=4pt,bottom=4pt}

% EU emblem (12 gold stars on blue field, aspect ratio 3:2), drawn in TikZ
% so no image path is needed.  Usage: \euemblem{<width in cm>}
\usepackage{tikz}
\usetikzlibrary{shapes.geometric}
\definecolor{eublue}{RGB}{0,51,153}
\definecolor{eustars}{RGB}{255,204,0}
\newcommand{\euemblem}[1]{%
\begin{tikzpicture}[x=#1/3,y=#1/3]
\fill[eublue] (0,0) rectangle (3,2);
\foreach \i in {0,...,11} {%
  \node[star, star points=5, star point ratio=2.618,
        fill=eustars, draw=eustars, inner sep=0,
        minimum size=#1/13] at ({1.5+0.65*cos(\i*30+90)},{1+0.65*sin(\i*30+90)}) {};%
}
\end{tikzpicture}%
}
