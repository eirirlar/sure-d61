% Figures — prefer near-reference placement but allow float when needed
% to avoid whitespace-heavy pages. Original constraint of [H] (exact
% position) produced blank space when a figure didn't fit; [!ht] tries
% here-or-top with high priority, falling through to bottom or float page
% only when nothing else works.
\usepackage{float}
\floatplacement{figure}{!ht}

% Running footer with project + deliverable + date + page number.

% TikZ for the cover-page image overlay and the EU emblem.
\usepackage{tikz}
\usetikzlibrary{shapes.geometric}

% EU emblem (12 gold stars on blue field, aspect ratio 3:2).
% Usage: \euemblem{<width in cm>}
\definecolor{eublue}{RGB}{0,51,153}
\definecolor{eustars}{RGB}{255,204,0}
\newcommand{\euemblem}[1]{%
\begin{tikzpicture}[x=#1/3,y=#1/3]
\fill[eublue] (0,0) rectangle (3,2);
\foreach \i in {0,...,11} {%
  \node[star, star points=5, star point ratio=2.618,
        fill=eustars, draw=eustars, inner sep=0,
        minimum size=#1/9] at ({1.5+0.65*cos(\i*30+90)},{1+0.65*sin(\i*30+90)}) {};%
}
\end{tikzpicture}%
}

% Named colours (NavyBlue etc) via svgnames option of xcolor.
% pandoc loads xcolor already; add svgnames.
